# 225. Implement Stack using Queues

## MySolution

class MyStack {

    ArrayDeque<Integer> q1, q2;

    public MyStack() {
        q1 = new ArrayDeque<>();
        q2 = new ArrayDeque<>();     
    }
    
    public void push(int x) {
        if (q1.isEmpty() && q2.isEmpty()) {
            q1.addLast(x);
        } else if (!q1.isEmpty()) {
            q1.addLast(x);
        } else {
            q2.addLast(x);
        }
    }
    
    public int pop() {
        int val = 0;
        if (q2.isEmpty()) {
            while (q1.size() > 1) {
                q2.addLast(q1.removeFirst());
            }
            val = q1.removeFirst();
        } else {
            while (q2.size() > 1) {
                q1.addLast(q2.removeFirst());
            }
            val = q2.removeFirst();
        }
        return val;
    }
    
    public int top() {
        int val = 0;
        if (q2.isEmpty()) {
            while (q1.size() > 1) {
                q2.addLast(q1.removeFirst());
            }
            val = q1.removeFirst();
            q2.addLast(val);
        } else {
            while (q2.size() > 1) {
                q1.addLast(q2.removeFirst());
            }
            val = q2.removeFirst();
            q1.addLast(val);
        }
        return val;
    }
    
    public boolean empty() {
        return q1.isEmpty() && q2.isEmpty();
    }
}
