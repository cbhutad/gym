# 682. Baseball Game

## My Solution

``` java

class Solution {
    public int calPoints(String[] operations) {
        LinkedList<Integer> record = new LinkedList<>();
        int sumScore = 0;

        for (String operation : operations) {
            switch (operation) {
                case "+":
                    int num1 = record.pop();
                    int num2 = record.pop();
                    record.push(num2);
                    record.push(num1);
                    record.push(num1 + num2);
                    break;
                case "D":
                    int num = record.peek();
                    record.push(num * 2);
                    break;
                case "C":
                    record.pop();
                    break;
                default:
                    record.push(Integer.parseInt(operation));
                    break;
            }
        }

        while (!record.isEmpty()) {
            sumScore += record.pop();
        }

        return sumScore;
    }
}
```
