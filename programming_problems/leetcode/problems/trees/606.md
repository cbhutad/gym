# 606. Construct String from Binary Tree

## MySolution

``` java

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    StringBuilder sb = new StringBuilder("");

    public String tree2str(TreeNode root) {
        preorderCustom(root);
        return sb.toString();
    }

    private void preorderCustom(TreeNode root) {
        if (root == null) {
            return;
        }
        if (root.right == null && root.left == null) {
            sb.append(root.val);
            return;
        }
        sb.append(root.val);
        sb.append("(");
        preorderCustom(root.left);
        sb.append(")");
        if (root.right != null) {
            sb.append("(");
            preorderCustom(root.right);
            sb.append(")");
        }
    }
}
```
